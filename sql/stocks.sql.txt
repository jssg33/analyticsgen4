CREATE TABLE SystemStocks (
Id INTEGER PRIMARY KEY AUTOINCREMENT,
Ticker TEXT,
CompanyName TEXT,
Source INTEGER NOT NULL DEFAULT 0,
Country TEXT DEFAULT 'US',
PrimaryExchange TEXT,
SecondaryExchange TEXT
);
 

CREATE INDEX IX_SystemStocks_Ticker
ON SystemStocks (Ticker);
 
CREATE INDEX IX_SystemStocks_Source
ON SystemStocks (Source);
 
CREATE INDEX IX_SystemStocks_PrimaryExchange
ON SystemStocks (PrimaryExchange);
